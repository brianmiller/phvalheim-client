    static readonly Func<string, DiskSpace.Measurement> enough = p => new(Path.GetPathRoot(p), 100_000_000);

    static void Check(bool ok, string name, Outcome? outcome = null, string? details = null)
    {
        report.WriteLine((ok ? "PASS: " : "FAIL: ") + name);
        if (ok) passed++; else failed++;
        if (!ok && details != null) FixtureDetail("observations", details);
        if (!ok && outcome != null)
        {
            FixtureDetail("value", JsonSerializer.Serialize(outcome.Value));
            FixtureDetail("exception", outcome.Error?.ToString() ?? "<none>");
            if (outcome.Error is DiskSpace.InsufficientDiskSpaceException space)
                FixtureDetail("space", JsonSerializer.Serialize(space.Assessment));
            FixtureDetail("output", outcome.Output);
            FixtureDetail("log", outcome.Log);
        }
    }
    static void FixtureDetail(string label, string text)
    {
        string safe = DiagnosticLog.Redact(text.Replace(credential, "[REDACTED]"));
        if (safe.Length > 4096) safe = safe[..2048] + "\n[truncated]\n" + safe[^2048..];
        report.WriteLine("FIXTURE " + label + ": " + JsonSerializer.Serialize(safe));
    }
    static void Put(string path, string value)
    {
        Directory.CreateDirectory(Path.GetDirectoryName(path)!);
        File.WriteAllText(path, value);
    }
    static string Log() => File.ReadAllText(Path.Combine(logRoot, "logs", "phvalheim-client.log"));
    static string Flat(string text) => Regex.Replace(text, @"\s+", "");
    static bool Has(string text, string value) => Flat(text).Contains(Flat(value), StringComparison.Ordinal);
    static object? Invoke(Type type, string name, object?[] values, Func<string, DiskSpace.Measurement>? measure = null)
    {
        var method = type.GetMethods(methods).Where(m => m.Name == name)
            .Where(m => m.GetParameters().Length == values.Length ||
                (m.GetParameters().Length == values.Length + 1 &&
                 m.GetParameters()[^1].ParameterType == typeof(Func<string, DiskSpace.Measurement>)))
            .OrderByDescending(m => m.GetParameters().Length).First();
        var args = method.GetParameters().Length == values.Length ? values : values.Concat(new object?[] { measure }).ToArray();
        try { return method.Invoke(null, args); }
        catch (TargetInvocationException ex) { throw ex.InnerException!; }
    }
    sealed record Outcome(object? Value, Exception? Error, string Output, string Log);
    static Outcome Capture(Func<object?> action)
    {
        int offset = Log().Length;
        using var output = new StringWriter();
        Console.SetOut(output); Console.SetError(output);
        object? value = null; Exception? error = null;
        try { value = action(); } catch (Exception ex) { error = ex; }
        finally { Console.SetOut(report); Console.SetError(report); }
        captured.Append(output);
        return new(value, error, output.ToString(), Log()[offset..]);
    }
    static void Run(string name, Action action)
    {
        try { action(); }
        catch (Exception ex)
        {
            Check(false, name + "_HARNESS_EXCEPTION_" + ex.GetType().Name);
            FixtureDetail("exception", ex.ToString());
        }
    }
    static void ErrorEvidence(Outcome outcome, string prefix, string path, string type)
    {
        Check(Has(outcome.Output, path) && outcome.Output.Contains(type), prefix + "_OUTPUT_PATH_AND_TYPE", outcome);
        Check(outcome.Log.Contains(path) && outcome.Log.Contains(type) && outcome.Log.Contains("HResult=0x")
            && outcome.Log.Contains("   at "), prefix + "_FULL_EXCEPTION_LOGGED", outcome);
        Check(Regex.Matches(outcome.Output, "Diagnostic log:").Count == 1, prefix + "_ONE_SUPPORT_LINE", outcome);
        Check(!outcome.Output.Contains(credential) && !outcome.Log.Contains(credential)
            && outcome.Output.Contains("ERROR:"), prefix + "_CREDENTIAL_ABSENT_ERROR_PRESENT", outcome);
    }
    static void CollisionEvidence(Outcome outcome, string prefix, string path)
    {
        var type = Regex.Match(outcome.Log, @"System\.(?:IO\.)?(UnauthorizedAccessException|IOException)\b");
        Check(type.Success && Has(outcome.Output, "existing directory occupies the expected file path"),
            prefix + "_DIRECTORY_COLLISION_MEANING", outcome);
        ErrorEvidence(outcome, prefix, path, type.Success ? type.Groups[1].Value : "<missing exception type>");
    }
    static Exception Denied(string path)
    {
        try { File.WriteAllText(path, "must not overwrite directory"); }
        catch (Exception ex) when (ex is IOException or UnauthorizedAccessException) { return ex; }
        throw new Exception("Directory-as-file fixture did not deny the write.");
    }
    static Exception DeniedCopy(string source, string target)
    {
        try { File.Copy(source, target, true); }
        catch (Exception ex) when (ex is IOException or UnauthorizedAccessException) { return ex; }
        throw new Exception("Directory-as-file fixture did not deny the copy.");
    }
    static void Backend()
    {
        string legacyRoot = OperatingSystem.IsWindows()
            ? Environment.ExpandEnvironmentVariables(@"%appdata%\PhValheim\worlds")
            : Environment.GetFolderPath(Environment.SpecialFolder.UserProfile) + "/.config/PhValheim/worlds";
        string isolatedWorldParent = work;
        string phvalheimHostNoPort = Path.GetRelativePath(legacyRoot, isolatedWorldParent);
        const string worldName = "test";
        string world = Path.Combine(isolatedWorldParent, worldName, worldName);
        string target = Path.Combine(world, "BepInEx", "phvalheim.backend");
        string oldBepInExRoot = OperatingSystem.IsWindows()
            ? Environment.ExpandEnvironmentVariables(@"%appdata%\PhValheim\worlds\" + phvalheimHostNoPort
                + "\\" + worldName + "\\" + worldName + @"\BepInEx\")
            : legacyRoot + "/" + phvalheimHostNoPort + "/" + worldName + "/" + worldName + "/BepInEx/";
        string oldTarget = Environment.ExpandEnvironmentVariables(oldBepInExRoot + "phvalheim.backend");
        string normalizedOldTarget = Path.GetFullPath(oldTarget);
        var comparison = OperatingSystem.IsWindows() ? StringComparison.OrdinalIgnoreCase : StringComparison.Ordinal;
        if (Path.IsPathRooted(phvalheimHostNoPort)
            || !normalizedOldTarget.Equals(Path.GetFullPath(target), comparison)
            || !normalizedOldTarget.StartsWith(Path.TrimEndingDirectorySeparator(isolatedWorldParent)
                + Path.DirectorySeparatorChar, comparison))
            throw new Exception("Refusing a legacy backend target outside the isolated fixture.");
        Check(true, "BACKEND_LEGACY_TARGET_EQUALS_ISOLATED_FIXTURE");
        PhValheim.Platform.State.PhValheimServerWorld = world;
        Directory.CreateDirectory(target);
        Exception stimulus = Denied(target);
        Check(stimulus is IOException or UnauthorizedAccessException, "BACKEND_ACTUAL_DENIED_WRITE_" + stimulus.GetType().Name);
        var result = Capture(() => PhValheim.Prep.PhValheim.WriteBackendFile(worldName, phvalheimHostNoPort, "http://localhost/" + credential));
        Check(result.Error == null && result.Value is false, "BACKEND_RETURNS_FALSE_WITHOUT_THROW", result);
        CollisionEvidence(result, "BACKEND", target);
        Check(Directory.Exists(target), "BACKEND_COLLISION_DIRECTORY_PRESERVED");
        if (OperatingSystem.IsWindows())
            Check(result.Log.Contains("Owner SID:") && result.Log.Contains("ACL:"), "BACKEND_WINDOWS_OWNER_ACL_EVIDENCE");
        Directory.Move(target, target + ".collision");
        var success = Capture(() => PhValheim.Prep.PhValheim.WriteBackendFile(worldName, phvalheimHostNoPort, "http://localhost/control"));
        Check(success.Value is true && File.ReadAllText(target) == "http://localhost/control", "BACKEND_SUCCESS_CONTROL");
    }
    static void Copy()
    {
        string source = Path.Combine(root, "copy-source"), destination = Path.Combine(root, "copy-target");
        Put(Path.Combine(source, "nested", "loader.so"), "loader-control");

        Check(result.Error is IOException or UnauthorizedAccessException, "COPY_PROPAGATES_ACCESS_FAILURE");
        Check(result.Error?.Data["PhValheim.Path"] as string == target, "COPY_EXACT_NESTED_TARGET_METADATA");
        Check(result.Error != null && result.Log.Contains(result.Error.GetType().FullName!)
            && result.Log.Contains(target) && result.Log.Contains("HResult=0x") && result.Log.Contains("   at ")
            && result.Log.Contains("existing directory occupies the expected file path"), "COPY_EXACT_NESTED_TARGET_LOGGED", result);
        Directory.Move(target, target + ".collision");
        var success = Capture(() => Invoke(typeof(PhValheim.Tooling.PhValheim), "CloneDirectory", new object?[] { source, destination }, enough));
        Check(success.Error == null && File.ReadAllText(target) == "loader-control", "COPY_SUCCESS_CONTROL");
        if (!baseline)
        {
            Put(target, "existing-copy-control");
            var low = Capture(() => Invoke(typeof(PhValheim.Tooling.PhValheim), "CopyFile",
                new object?[] { Path.Combine(source, "nested", "loader.so"), target }, p => new(Path.GetPathRoot(p), 0)));
            Check(low.Error is DiskSpace.InsufficientDiskSpaceException && File.ReadAllText(target) == "existing-copy-control",
                "COPY_LOW_SPACE_EXISTING_TARGET_BYTES_INTACT", low);
            Check(low.Log.Contains("available bytes: 0") && low.Log.Contains(target)
                && !Directory.GetFiles(Path.GetDirectoryName(target)!, ".phvalheim-*.tmp").Any(),
                "COPY_LOW_SPACE_EVIDENCE_NO_STAGING", low);
        }
    }
    static Outcome WriteNative(bool backend, string source, string target, string contents) => Capture(() => backend
        ? Invoke(typeof(PhValheim.Prep.PhValheim), "WriteBackendFile", new object?[] { target, contents })
        : Invoke(typeof(PhValheim.Tooling.PhValheim), "CopyFile", new object?[] { source, target }, enough));
    static bool WriteSucceeded(Outcome result, bool backend) => result.Error == null && (!backend || result.Value is true);
    static bool NoWriteTemps(string folder) => !Directory.GetFiles(folder, ".phvalheim-*.tmp").Any();
    static void NativeFiles()
    {
        bool windows = OperatingSystem.IsWindows(), mac = OperatingSystem.IsMacOS(), linux = OperatingSystem.IsLinux();
        if (!windows && !mac && !linux)
        {
            report.WriteLine("LIMITS: native file tests require Windows, macOS or Linux.");
            Check(false, "NATIVE_FILES_UNSUPPORTED_OS");
            return;
        }
        foreach (string name in new[] { "doorstop_config.ini", "winhttp.dll", "phvalheim.backend" })
        {
            bool backend = name == "phvalheim.backend";
            string prefix = (windows ? "WINDOWS" : mac ? "MAC" : "LINUX") + "_" + name.Replace('.', '_').ToUpperInvariant();
            string folder = Path.Combine(root, "native-files", name);
            string target = Path.Combine(folder, (linux ? "." : "") + name);
            string source = Path.Combine(folder, ".source");
            const string contents = "native-new-content-雪";
            Put(source, contents);
            Run(prefix, () => {
                var created = WriteNative(backend, source, target, contents);
                Check(WriteSucceeded(created, backend) && File.ReadAllText(target) == contents,
                    prefix + "_NEW_FILE_CONTROL", created);
                Put(target, "plain-old-control");
                var plain = WriteNative(backend, source, target, contents);
                Check(WriteSucceeded(plain, backend) && File.ReadAllText(target) == contents,
                    prefix + "_ORDINARY_OVERWRITE_CONTROL", plain);
                Put(target, "hidden-old-control");
                FileAttributes original = File.GetAttributes(target);
                try
                {
                    if (!linux) File.SetAttributes(target, original | FileAttributes.Hidden);
                    bool hidden = (File.GetAttributes(target) & FileAttributes.Hidden) != 0;
                    Check(hidden, prefix + "_ACTUAL_HIDDEN_FIXTURE");
                    if (!hidden)
                    {
                        report.WriteLine("LIMITS: filesystem did not retain Hidden; native Hidden coverage unavailable for " + prefix);
                        return;
                    }
                    var overwritten = WriteNative(backend, source, target, contents);
                    Check(WriteSucceeded(overwritten, backend)
                        && File.ReadAllBytes(target).SequenceEqual(Encoding.UTF8.GetBytes(contents)),
                        prefix + "_HIDDEN_OVERWRITE_BYTES", overwritten);
                    Check((File.GetAttributes(target) & FileAttributes.Hidden) != 0,
                        prefix + "_HIDDEN_RETAINED", overwritten);
                    Check(Directory.GetFiles(folder).Select(Path.GetFileName).OrderBy(n => n, StringComparer.Ordinal)
                        .SequenceEqual(new[] { Path.GetFileName(source), Path.GetFileName(target) }.OrderBy(n => n, StringComparer.Ordinal))
                        && File.ReadAllText(source) == contents && NoWriteTemps(folder),
                        prefix + "_NAMES_SOURCE_BYTES_NO_TEMP_LEFTOVERS", overwritten);
                    if (windows) ReadOnlyHidden(backend, source, target, contents, prefix);
                }
                finally { File.SetAttributes(target, original); }
            });
            if (windows) Run(prefix + "_DACL", () => DenyWriteData(backend, source, Path.Combine(folder, "acl-target"), contents, prefix));
        }
        if (windows) Run("WINDOWS_STAGING", WriterThrows);
        else report.WriteLine("LIMITS: Windows Hidden/read-only/DACL/staging tests require native Windows; Windows helper is not invoked on Unix.");
        if (!mac) report.WriteLine("LIMITS: macOS Hidden-flag preservation requires native macOS; Linux dotfile coverage does not establish it.");
    }
    static void DenialEvidence(Outcome result, bool backend, string target, string prefix)
    {
        if (backend)
        {
            Check(result.Error == null && result.Value is false, prefix + "_RETURNS_FALSE", result);
            ErrorEvidence(result, prefix, target, nameof(UnauthorizedAccessException));
        }
        else
        {
            Check(result.Error is UnauthorizedAccessException && result.Error.HResult == unchecked((int)0x80070005),
                prefix + "_ACTUAL_ACCESS_DENIED", result);
            Check(result.Error?.Data["PhValheim.Path"] as string == target
                && result.Log.Contains(typeof(UnauthorizedAccessException).FullName!)
                && result.Log.Contains(target) && result.Log.Contains("HResult=0x80070005") && result.Log.Contains("   at "),
                prefix + "_ACTUAL_TYPE_PATH_AND_STACK", result);
        }
    }
    static void ReadOnlyHidden(bool backend, string source, string target, string contents, string prefix)
    {
        File.SetAttributes(target, File.GetAttributes(target) & ~FileAttributes.Hidden);
        Put(target, "readonly-original-control");
        File.SetAttributes(target, File.GetAttributes(target) | FileAttributes.Hidden | FileAttributes.ReadOnly);
        byte[] before = File.ReadAllBytes(target);
        FileAttributes attributes = File.GetAttributes(target);
        Check((attributes & (FileAttributes.Hidden | FileAttributes.ReadOnly)) == (FileAttributes.Hidden | FileAttributes.ReadOnly),
            prefix + "_READONLY_HIDDEN_FIXTURE");
        var denied = WriteNative(backend, source, target, contents);
        DenialEvidence(denied, backend, target, prefix + "_READONLY");
        Check(File.ReadAllBytes(target).SequenceEqual(before) && File.GetAttributes(target) == attributes
            && NoWriteTemps(Path.GetDirectoryName(target)!), prefix + "_READONLY_BYTES_ATTRS_INTACT", denied);
    }
    static string SecurityState(string target) => new FileInfo(target)
        .GetAccessControl(AccessControlSections.Access | AccessControlSections.Owner | AccessControlSections.Group)
        .GetSecurityDescriptorSddlForm(AccessControlSections.Access | AccessControlSections.Owner | AccessControlSections.Group);
    static void DenyWriteData(bool backend, string source, string target, string contents, string prefix)
    {
        if (!OperatingSystem.IsWindows()) throw new PlatformNotSupportedException("DACL fixture requires Windows.");
        Put(target, "dacl-original-control");
        FileAttributes originalAttributes = File.GetAttributes(target);
        FileSecurity? original = null;
        string? originalSecurity = null;
        bool changed = false;
        try
        {
            original = new FileInfo(target).GetAccessControl(AccessControlSections.Access);
            originalSecurity = SecurityState(target);
            using var identity = WindowsIdentity.GetCurrent();
            var rule = new FileSystemAccessRule(identity.User ?? throw new InvalidOperationException("No Windows user SID."),
                FileSystemRights.WriteData, AccessControlType.Deny);
            var security = new FileInfo(target).GetAccessControl(AccessControlSections.Access);
            security.AddAccessRule(rule);
            File.SetAttributes(target, originalAttributes | FileAttributes.Hidden);
            changed = true;
            new FileInfo(target).SetAccessControl(security);
            string deniedSecurity = SecurityState(target);
            byte[] before = File.ReadAllBytes(target);
            FileAttributes attributes = File.GetAttributes(target);
            Check(deniedSecurity != originalSecurity && (attributes & FileAttributes.Hidden) != 0,
                prefix + "_DACL_HIDDEN_FIXTURE");
            var probe = Capture(() => {
                using var stream = new FileStream(target, FileMode.Open, FileAccess.Write, FileShare.None);
                return null;
            });
            Check(probe.Error is UnauthorizedAccessException && probe.Error.HResult == unchecked((int)0x80070005),
                prefix + "_DACL_OS_DENIES_WRITE_DATA", probe);
            if (probe.Error is not UnauthorizedAccessException)
            {
                report.WriteLine("LIMITS: DACL fixture did not produce actual Windows access denial for " + prefix);
                return;
            }
            var denied = WriteNative(backend, source, target, contents);
            DenialEvidence(denied, backend, target, prefix + "_DACL");
            Check(File.ReadAllBytes(target).SequenceEqual(before) && File.GetAttributes(target) == attributes
                && SecurityState(target) == deniedSecurity && NoWriteTemps(Path.GetDirectoryName(target)!),
                prefix + "_DACL_BYTES_ATTRS_ACL_INTACT", denied);
        }
        catch (Exception ex) when (ex is PlatformNotSupportedException or NotSupportedException or UnauthorizedAccessException)
        {
            report.WriteLine("LIMITS: native DACL coverage unavailable or failed: " + prefix + "; " + ex.GetType().Name);
            Check(false, prefix + "_DACL_UNVERIFIED");
            FixtureDetail("exception", ex.ToString());
        }
        finally
        {
            try
            {
                if (changed && original != null)
                {
                    var restore = new FileSecurity();
                    restore.SetSecurityDescriptorBinaryForm(original.GetSecurityDescriptorBinaryForm(), AccessControlSections.Access);
                    new FileInfo(target).SetAccessControl(restore);
                    Check(SecurityState(target) == originalSecurity, prefix + "_DACL_RESTORED");
                    using var stream = new FileStream(target, FileMode.Open, FileAccess.Write, FileShare.None);
                    Check(stream.CanWrite, prefix + "_DACL_RESTORED_WRITE_CONTROL");
                }
            }
            finally { File.SetAttributes(target, originalAttributes); }
        }
    }
    static void WriterThrows()
    {
        if (!OperatingSystem.IsWindows()) throw new PlatformNotSupportedException("Managed writer fixture requires Windows.");
        string folder = Path.Combine(root, "writer-throws"), target = Path.Combine(folder, "doorstop_config.ini");
        Put(target, "original-before-partial-staging");
        FileAttributes original = File.GetAttributes(target);
        File.SetAttributes(target, original | FileAttributes.Hidden);
        try
        {
            byte[] before = File.ReadAllBytes(target);
            FileAttributes attributes = File.GetAttributes(target);
            string security = SecurityState(target);
            string[] entries = Directory.GetFileSystemEntries(folder).OrderBy(p => p, StringComparer.Ordinal).ToArray();
            bool staged = false;
            var fault = new IOException("fixture-writer-threw-after-partial-staging");
            Action<Stream> writer = stream => {
                stream.Write(Encoding.UTF8.GetBytes("partial-stage"));
                stream.Flush();
                staged = stream.Length > 0 && Directory.GetFiles(folder, ".phvalheim-write-*.tmp").Length == 1;
                throw fault;
            };
            var result = Capture(() => Invoke(typeof(PhValheim.Tooling.PhValheim).Assembly.GetType("PhValheim.ManagedFile", true)!,
                "Write", new object?[] { target, writer }));
            Check(staged && ReferenceEquals(result.Error, fault), "WINDOWS_WRITER_PARTIAL_STAGE_ACTUAL_THROW", result);
            Check(File.ReadAllBytes(target).SequenceEqual(before) && File.GetAttributes(target) == attributes
                && SecurityState(target) == security, "WINDOWS_WRITER_THROW_ORIGINAL_BYTES_ATTRS_ACL_INTACT", result);
            Check(entries.SequenceEqual(Directory.GetFileSystemEntries(folder).OrderBy(p => p, StringComparer.Ordinal))
                && NoWriteTemps(folder), "WINDOWS_WRITER_THROW_NO_TEMP_LEFTOVERS", result);
        }
        finally { File.SetAttributes(target, original); }
    }

    static void Sync(Server server)
    {
        string world = Path.Combine(root, "sync", "test"), game = Path.Combine(root, "game");
        PhValheim.Platform.State.WorldName = "test";
        PhValheim.Platform.State.PhValheimServerWorld = world;
        PhValheim.Platform.State.PhValheimServerRoot = Path.GetDirectoryName(world)!;
        PhValheim.Platform.State.PhValheimDir = Path.Combine(root, "legacy");
        PhValheim.Platform.State.ValheimDir = game;
        Directory.CreateDirectory(Path.GetDirectoryName(world)!);
        Directory.CreateDirectory(game);
        string[] foreign = OperatingSystem.IsWindows()
            ? new[] { Path.Combine("doorstop_libs", "libdoorstop_x64.so"), Path.Combine("doorstop_libs", "libdoorstop.dylib") }
            : OperatingSystem.IsLinux()
                ? new[] { "winhttp.dll", Path.Combine("doorstop_libs", "libdoorstop.dylib") }
                : new[] { "winhttp.dll", "doorstop_config.ini", Path.Combine("doorstop_libs", "libdoorstop_x64.so"),
                    Path.Combine("doorstop_libs", "libdoorstop.dylib") };
        if (!baseline)
            foreach (string file in foreign) Put(Path.Combine(game, file), "foreign-sentinel:" + file);
        string blocked = Path.Combine(game, "doorstop_libs", "nested");
        if (!baseline) Put(blocked, "foreign-blocked-directory-control");
        void ForeignIntact(string suffix)
        {
            if (baseline) return;
            Check(foreign.All(file => File.Exists(Path.Combine(game, file))
                && File.ReadAllText(Path.Combine(game, file)) == "foreign-sentinel:" + file)
                && File.ReadAllText(blocked) == "foreign-blocked-directory-control",
                "SYNC_FOREIGN_LOADERS_UNTOUCHED_" + suffix);
        }
        bool mac = OperatingSystem.IsMacOS() && !baseline;
        string relative = baseline ? Path.Combine("doorstop_libs", "nested", "loader.so")
            : OperatingSystem.IsWindows() ? "winhttp.dll" : Path.Combine("doorstop_libs", "libdoorstop_x64.so");
        string target = Path.Combine(game, relative);
        string expected = baseline ? "loader-control" : OperatingSystem.IsWindows() ? "windows-loader" : "linux-loader";
        string[] before = GameSnapshot(game);
        if (!mac)
        {
            string source = Path.Combine(root, "sync-copy-source", "loader.so");
            Put(source, expected);
            Directory.CreateDirectory(target);
            Exception stimulus = DeniedCopy(source, target);
            Check(stimulus is IOException or UnauthorizedAccessException, "SYNC_ACTUAL_DENIED_COPY_" + stimulus.GetType().Name);
        }
        var result = Capture(() => Invoke(typeof(PhValheim.Syncer.PhValheim), "Sync", new object?[] { server.Url }, enough));
        if (mac)
        {
            Check(result.Error == null && result.Value is true && server.ZipRequests > 0
                && File.ReadAllText(Path.Combine(world, "doorstop_libs", "libdoorstop.dylib")) == "mac-loader",
                "SYNC_MAC_REAL_HTTP_WORLD_DYLIB_SUCCESS", result);
            Check(before.SequenceEqual(GameSnapshot(game)), "SYNC_MAC_NO_GAME_COPY", result);
        }
        else
        {
            Check(result.Error == null && result.Value is false && server.ZipRequests > 0
                && File.Exists(Path.Combine(world, "doorstop_config.ini")), "SYNC_REAL_HTTP_AND_COPY_FAILURE", result);
            CollisionEvidence(result, "SYNC", target);
            Check(!result.Output.Contains("usually antivirus", StringComparison.OrdinalIgnoreCase)
                && result.Output.Contains("ERROR:"), "SYNC_NO_UNSUPPORTED_ANTIVIRUS_BLAME", result);
            Directory.Move(target, target + ".collision");
        }

    // Raw loopback HTTP permits a deterministic short body and unknown Content-Length without URL ACL setup.
    sealed class Server : IDisposable
    {
        public const string Payload = "real-http-payload-control";
        readonly TcpListener listener = new(IPAddress.Loopback, 0);
        readonly CancellationTokenSource stop = new();
        readonly Task serving;
        readonly byte[] archive;
        readonly string md5;
        public readonly HashSet<string> Routes = new();
        public int ZipRequests;
        public string Url { get; }
        public Server(string path)
        {
            archive = File.ReadAllBytes(path); md5 = PhValheim.Tooling.PhValheim.getMD5(path);
            listener.Start(); Url = "http://127.0.0.1:" + ((IPEndPoint)listener.LocalEndpoint).Port;
            serving = Task.Run(Serve);
        }
        async Task Serve()
        {
            try
            {
                while (!stop.IsCancellationRequested)
                {
                    using var client = await listener.AcceptTcpClientAsync(stop.Token);
                    using var stream = client.GetStream();
                    using var reader = new StreamReader(stream, Encoding.ASCII, false, 1024, true);
                    string request = await reader.ReadLineAsync(stop.Token) ?? "";
                    while (!string.IsNullOrEmpty(await reader.ReadLineAsync(stop.Token))) { }
                    string route = request.Split(' ')[1].Split('?')[0].TrimStart('/');
                    lock (Routes) Routes.Add(route);
                    byte[] body = Encoding.UTF8.GetBytes(Payload);
                    string status = "200 OK", length;
                    if (route.EndsWith(".zip")) { Interlocked.Increment(ref ZipRequests); body = archive; }
                    else if (route == "api.php") body = Encoding.UTF8.GetBytes("world=" + md5 + "\nmods=fixture-mods\nconfig=" + md5 + "\n");
                    else if (route == "httpfail") { status = "503 Service Unavailable"; body = Encoding.UTF8.GetBytes("fixture-http-failure"); }
                    length = route == "unknown" ? "" : "Content-Length: " + (body.Length + (route == "truncated" ? 100 : 0)) + "\r\n";
                    byte[] header = Encoding.ASCII.GetBytes("HTTP/1.1 " + status + "\r\n" + length + "Connection: close\r\n\r\n");
                    try { await stream.WriteAsync(header, stop.Token); await stream.WriteAsync(body, stop.Token); }
                    catch (IOException) { }
                }
            }
            catch (OperationCanceledException) { }
            catch (SocketException) when (stop.IsCancellationRequested) { }
        }
        public void Dispose()
        {
            stop.Cancel(); listener.Stop();
            if (!serving.Wait(TimeSpan.FromSeconds(2))) throw new Exception("HTTP fixture failed to stop.");
            stop.Dispose();
        }
    }
